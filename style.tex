\newcommand{\maintitle}[1]{%
    \par\vspace{12pt}%      <- Mantiene la distancia con la sección anterior
    {\bfseries\fontsize{8.8pt}{10.5pt}\selectfont\color{darktext}\uppercase{#1}}\\[-4pt]%
    {\color{darktext}\rule{\linewidth}{1pt}}\par
    \vspace{-4pt}%          <- Espacio negativo para acercar de inmediato el contenido de abajo
}

% Modificación en style.tex para espaciar y destacar cada entrada

\newcommand{\timelineentry}[3]{%
    \vspace{4.5pt}% Espacio previo que desvincula la entrada de la anterior
    \noindent
    \begin{tikzpicture}[baseline=(entrycontent.north)]
        \node[inner sep=0pt, outer sep=0pt, text width=\linewidth-14pt, align=left] (entrycontent) {
            {\fontsize{8.8pt}{11pt}\selectfont\textbf{\color{darktext} #1}} \hfill {\scriptsize\color{lightgray} #2}\\[2.5pt]
            #3
        };
        % Línea vertical del timeline con mayor contraste
        \draw[accentblue!60, line width=1.2pt] ([xshift=-7pt, yshift=-3.5pt]entrycontent.north west) -- ([xshift=-7pt, yshift=2pt]entrycontent.south west);
        \fill[accentblue] ([xshift=-7pt, yshift=-3.5pt]entrycontent.north west) circle (2.3pt);
    \end{tikzpicture}\par\vspace{3.5pt}%
}
